# OMPI MCA framework base dirs + statically-linked components (Windows).
#
# All MCA sources are accumulated into OMPI_MCA_SOURCES (consumed by
# ompi/CMakeLists.txt when declaring the open-mpi target).
set(OMPI_MCA_SOURCES "")
set(OMPI_MCA_FRAMEWORKS "")

# ompi_add_mca_framework(<framework> [COMPONENTS c1 c2 ...])
function(ompi_add_mca_framework fw)
    cmake_parse_arguments(ARG "" "" "COMPONENTS" ${ARGN})
    list(APPEND OMPI_MCA_FRAMEWORKS ${fw})
    set(OMPI_MCA_FRAMEWORKS "${OMPI_MCA_FRAMEWORKS}" PARENT_SCOPE)

    # -- framework base sources --
    file(GLOB _base_srcs CONFIGURE_DEPENDS
         "${CMAKE_CURRENT_SOURCE_DIR}/${fw}/base/*.c")
    foreach(f IN LISTS _base_srcs)
        list(APPEND OMPI_MCA_SOURCES "${f}")
    endforeach()

    # -- static-components.h --
    set(_hdr_dir "${CMAKE_BINARY_DIR}/ompi/mca/${fw}/base")
    file(MAKE_DIRECTORY "${_hdr_dir}")
    set(_externs "")
    set(_entries "")
    foreach(comp IN LISTS ARG_COMPONENTS)
        string(APPEND _externs
            "extern const mca_base_component_t *mca_${fw}_${comp}_component_ptr;\n")
        string(APPEND _entries
            "    &mca_${fw}_${comp}_component_ptr,\n")
    endforeach()
    file(WRITE "${_hdr_dir}/static-components.h"
"/*
 * Generated by CMake -- lists the statically linked components of the
 * ompi ${fw} framework (mirrors config/ompi_mca.m4 output).
 */
#if defined(c_plusplus) || defined(__cplusplus)
extern \"C\" {
#endif

${_externs}
const mca_base_component_t **mca_${fw}_base_static_components[] = {
${_entries}    NULL
};

#if defined(c_plusplus) || defined(__cplusplus)
}
#endif
")

    # -- component sources --
    foreach(comp IN LISTS ARG_COMPONENTS)
        file(GLOB _comp_srcs CONFIGURE_DEPENDS
             "${CMAKE_CURRENT_SOURCE_DIR}/${fw}/${comp}/*.c")
        foreach(f IN LISTS _comp_srcs)
            if(NOT f MATCHES "(test|_main)\\.c$")
                list(APPEND OMPI_MCA_SOURCES "${f}")
            endif()
        endforeach()
    endforeach()

    set(OMPI_MCA_SOURCES "${OMPI_MCA_SOURCES}" PARENT_SCOPE)
endfunction()

# ------------------------------------------------------------------
# Windows component selection (curated)
#
# Goal for v1: MPI_Init + MPI_Send/Recv over TCP (pml/ob1 + btl tcp/self)
# plus enough collectives/IO to run the examples.  RDMA/UCX/OFI/Portals4
# components are all Unix-only; osc/rdma and osc/sm need btl rdma or smsc
# support and are deferred.
# ------------------------------------------------------------------
ompi_add_mca_framework(bml      COMPONENTS r2)
ompi_add_mca_framework(coll     COMPONENTS basic self tuned inter libnbc)
ompi_add_mca_framework(hook     COMPONENTS comm_method)
ompi_add_mca_framework(mtl)          # base only: no libfabric/psm on Windows
ompi_add_mca_framework(op)           # base only: generic C op implementations
ompi_add_mca_framework(osc)          # base only: rdma/sm deferred (need smsc/rdma btl)
ompi_add_mca_framework(part     COMPONENTS persist)
ompi_add_mca_framework(pml      COMPONENTS ob1)
ompi_add_mca_framework(topo     COMPONENTS basic)
ompi_add_mca_framework(vprotocol)    # base only: no checksum layer yet

# MPI-IO chain (io/ompio + fbtl/fcoll/fs/sharedfp) is deferred: the only
# portable fbtl component is aio-based (no POSIX AIO on Windows) and the
# others need it.  Frameworks stay compiled for structure; MPI_File_*
# will fail component selection until a Windows fbtl lands.
ompi_add_mca_framework(io)
ompi_add_mca_framework(fbtl)
ompi_add_mca_framework(fcoll)
ompi_add_mca_framework(fs)
ompi_add_mca_framework(sharedfp)

# propagate the accumulated source list up to ompi/CMakeLists.txt
set(OMPI_MCA_SOURCES "${OMPI_MCA_SOURCES}" PARENT_SCOPE)

# generate ompi/include/ompi/frameworks.h (autogen.pl equivalent)
set(_fw_decls "")
set(_fw_entries "")
foreach(fw IN LISTS OMPI_MCA_FRAMEWORKS)
    string(APPEND _fw_decls "OMPI_DECLSPEC extern mca_base_framework_t ompi_${fw}_base_framework;\n")
    string(APPEND _fw_entries "    &ompi_${fw}_base_framework,\n")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/ompi/include/ompi")
file(WRITE "${CMAKE_BINARY_DIR}/ompi/include/ompi/frameworks.h"
"/*
 * Generated by CMake -- mirrors autogen.pl's mca_generate_framework_header()
 */
#ifndef OMPI_FRAMEWORKS_H
#define OMPI_FRAMEWORKS_H

#include \"ompi_config.h\"
#include \"opal/mca/base/mca_base_framework.h\"

${_fw_decls}
static mca_base_framework_t *ompi_frameworks[] = {
${_fw_entries}    NULL
};

#endif /* OMPI_FRAMEWORKS_H */
")
