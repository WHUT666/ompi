# MCA framework base dirs + statically-linked components (Windows).

# All MCA sources are accumulated into OPAL_MCA_SOURCES (consumed by
# opal/CMakeLists.txt when declaring the open-pal target).
set(OPAL_MCA_SOURCES "")
# framework names accumulated for the generated opal/frameworks.h
set(OPAL_MCA_FRAMEWORKS "")

# opal_add_mca_framework(<framework> [COMPONENTS c1 c2 ...])
#
# Adds base/*.c of the framework, generates
# <build>/opal/mca/<fw>/base/static-components.h listing the selected
# static components, and adds each component's *.c sources.
function(opal_add_mca_framework fw)
    cmake_parse_arguments(ARG "" "" "COMPONENTS" ${ARGN})
    list(APPEND OPAL_MCA_FRAMEWORKS ${fw})
    set(OPAL_MCA_FRAMEWORKS "${OPAL_MCA_FRAMEWORKS}" PARENT_SCOPE)

    # -- framework base sources --
    file(GLOB _base_srcs CONFIGURE_DEPENDS
         "${CMAKE_CURRENT_SOURCE_DIR}/${fw}/base/*.c")
    foreach(f IN LISTS _base_srcs)
        list(APPEND OPAL_MCA_SOURCES "${f}")
    endforeach()

    # -- static-components.h --
    set(_hdr_dir "${CMAKE_BINARY_DIR}/opal/mca/${fw}/base")
    file(MAKE_DIRECTORY "${_hdr_dir}")
    set(_externs "")
    set(_entries "")
    foreach(comp IN LISTS ARG_COMPONENTS)
        string(APPEND _externs
            "extern const mca_base_component_t *mca_${fw}_${comp}_component_ptr;\n")
        string(APPEND _entries
            "    &mca_${fw}_${comp}_component_ptr,\n")
    endforeach()
    file(WRITE "${_hdr_dir}/static-components.h"
"/*
 * Generated by CMake -- lists the statically linked components of the
 * ${fw} framework (mirrors config/opal_mca.m4 output).
 */
#if defined(c_plusplus) || defined(__cplusplus)
extern \"C\" {
#endif

${_externs}
const mca_base_component_t **mca_${fw}_base_static_components[] = {
${_entries}    NULL
};

#if defined(c_plusplus) || defined(__cplusplus)
}
#endif
")

    # -- component sources --
    foreach(comp IN LISTS ARG_COMPONENTS)
        file(GLOB _comp_srcs CONFIGURE_DEPENDS
             "${CMAKE_CURRENT_SOURCE_DIR}/${fw}/${comp}/*.c")
        foreach(f IN LISTS _comp_srcs)
            # skip component-internal main()/test drivers if any
            if(NOT f MATCHES "(test|_main)\\.c$")
                list(APPEND OPAL_MCA_SOURCES "${f}")
            endif()
        endforeach()
    endforeach()

    set(OPAL_MCA_SOURCES "${OPAL_MCA_SOURCES}" PARENT_SCOPE)
endfunction()

# ------------------------------------------------------------------
# Windows component selection (curated)
#
# Frameworks with no components listed still get their base/ plumbing
# compiled plus an empty static-components.h.
# ------------------------------------------------------------------
opal_add_mca_framework(accelerator COMPONENTS null)
opal_add_mca_framework(allocator   COMPONENTS basic bucket)
opal_add_mca_framework(backtrace   COMPONENTS execinfo)
opal_add_mca_framework(btl        COMPONENTS self tcp)
opal_add_mca_framework(dl         COMPONENTS dlopen)  # dlfcn shim -> LoadLibrary
opal_add_mca_framework(hwloc)         # external hwloc build
opal_add_mca_framework(if         COMPONENTS bsdx_ipv4)   # getifaddrs shim
# env must be listed before config: base_open copies each field from the
# first component that provides it, so env vars only win if env comes first.
opal_add_mca_framework(installdirs COMPONENTS env config)
opal_add_mca_framework(memchecker)
opal_add_mca_framework(memcpy)
opal_add_mca_framework(memory)
opal_add_mca_framework(mpool)
opal_add_mca_framework(patcher)
opal_add_mca_framework(pmix)          # Phase 4: external pmix
opal_add_mca_framework(rcache)
opal_add_mca_framework(reachable  COMPONENTS weighted)
opal_add_mca_framework(shmem      COMPONENTS mmap)    # mmap shim -> CreateFileMapping
opal_add_mca_framework(smsc)          # TODO Phase 1c: windows component
opal_add_mca_framework(threads    COMPONENTS pthreads)
opal_add_mca_framework(timer)         # TODO Phase 1c: windows component

# propagate the accumulated source list up to opal/CMakeLists.txt
set(OPAL_MCA_SOURCES "${OPAL_MCA_SOURCES}" PARENT_SCOPE)

# generate opal/include/opal/frameworks.h (autogen.pl equivalent):
# declares every base framework object and the static registry array
set(_fw_decls "")
set(_fw_entries "")
foreach(fw IN LISTS OPAL_MCA_FRAMEWORKS)
    string(APPEND _fw_decls "OPAL_DECLSPEC extern mca_base_framework_t opal_${fw}_base_framework;\n")
    string(APPEND _fw_entries "    &opal_${fw}_base_framework,\n")
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/opal/include/opal/frameworks.h"
"/*
 * Generated by CMake -- mirrors autogen.pl's mca_generate_framework_header()
 * OPAL_DECLSPEC added vs upstream: framework objects live in open-pal and
 * must be dllexport'ed consistently with each base.h declaration.
 */
#ifndef OPAL_FRAMEWORKS_H
#define OPAL_FRAMEWORKS_H

#include \"opal_config.h\"
#include \"opal/mca/base/mca_base_framework.h\"

${_fw_decls}
static mca_base_framework_t *opal_frameworks[] = {
${_fw_entries}    NULL
};

#endif /* OPAL_FRAMEWORKS_H */
")
