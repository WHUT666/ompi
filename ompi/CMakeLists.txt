# libopen_mpi -- the internal OMPI library (all back-end code).
# The public MPI bindings library (mpi.dll) is Phase 6.

# ---- harvest core source lists from the per-dir Makefile.am files ----
# These files are all included into ompi/Makefile.am, so paths in them are
# relative to ompi/ (the ":ompi" prefix arg to the harvester).
# attribute..win are `include`d into ompi/Makefile.am (paths relative to
# ompi/); datatype and debuggers are real SUBDIRS (paths relative to their
# own directory).
set(_ompi_dirs
    attribute class communicator dpm errhandler file
    group info instance interlib message mpi op peruse proc request
    runtime tools win)
set(_ompi_subdirs datatype debuggers)

set(_harvest_includes "")
foreach(d IN LISTS _ompi_dirs)
    list(APPEND _harvest_includes --include "ompi/${d}/Makefile.am:ompi")
endforeach()
foreach(d IN LISTS _ompi_subdirs)
    list(APPEND _harvest_includes --include "ompi/${d}/Makefile.am:ompi/${d}")
endforeach()

execute_process(
    COMMAND ${Python3_EXECUTABLE}
            "${OMPI_SOURCE_DIR}/cmake/harvest_sources.py"
            --root "${OMPI_SOURCE_DIR}"
            --var OMPI_CORE_SOURCES
            --assign-pattern "libopen_mpi_la_SOURCES|libdatatype_la_SOURCES|libdebuggers_la_SOURCES"
            --exclude "peruse/|communicator/ft/|req_ft|mpiext|fortran|dlopen_test|predefined_gap_test|canary|msgq|ompi_common_dll"
            --out "${CMAKE_BINARY_DIR}/ompi_core_sources.cmake"
            ${_harvest_includes}
    RESULT_VARIABLE _harvest_rc
    OUTPUT_VARIABLE _harvest_out
    ERROR_VARIABLE  _harvest_err)
if(NOT _harvest_rc EQUAL 0)
    message(FATAL_ERROR "ompi source harvest failed: ${_harvest_err}")
endif()
include("${CMAKE_BINARY_DIR}/ompi_core_sources.cmake")

# ---- OMPI MCA frameworks/components ----
add_subdirectory(mca)

# ---- generated C bindings (Phase 6) ----
# On Windows the MPI/PMPI bindings are folded into open-mpi.dll rather than a
# separate libmpi: every ompi_* symbol the bindings need is defined there, and
# OMPI_BUILDING=1 exports the MPI_* entry points from the same DLL.
execute_process(
    COMMAND ${Python3_EXECUTABLE}
            "${OMPI_SOURCE_DIR}/cmake/gen_bindings.py"
            "${OMPI_SOURCE_DIR}" "${CMAKE_BINARY_DIR}"
            "${CMAKE_BINARY_DIR}/ompi/bindings"
    RESULT_VARIABLE _bind_rc
    OUTPUT_VARIABLE _bind_out
    ERROR_VARIABLE  _bind_err)
if(NOT _bind_rc EQUAL 0)
    message(FATAL_ERROR "bindings generation failed: ${_bind_err}")
endif()
include("${CMAKE_BINARY_DIR}/ompi/bindings/bindings_sources.cmake")

# ---- libopen_mpi ----
add_library(open-mpi SHARED
    ${OMPI_CORE_SOURCES}
    ${OMPI_MCA_SOURCES}
    ${BINDINGS_SOURCES}
    ${_ompi_bind_extra}
    ${_ompi_bind_misc}
    "${OMPI_SOURCE_DIR}/ompi/mpiext/mpiext.c")
# generated bindings use "#pragma weak" which MSVC doesn't know
set_source_files_properties(${BINDINGS_SOURCES} PROPERTIES
    COMPILE_OPTIONS "/wd4068")
target_link_options(open-mpi PRIVATE /DEBUG)

target_compile_definitions(open-mpi PRIVATE
    OMPI_BUILDING=1)

# MPI-IO (ompio) needs this private define
target_compile_definitions(open-mpi PRIVATE OMPI_MPIHANDLES_DLL_PREFIX="ompi_dbg_mpihandles")

# include order matters for mpi.h/ompi_config.h: the build dir must come
# first so the generated mpi.h wins, and ompi/include (source) provides
# ompi/constants.h etc.  win32fwd/shadow dirs come from the global
# include_directories().
target_include_directories(open-mpi PRIVATE
    "${CMAKE_BINARY_DIR}/ompi/include"    # generated mpi.h, ompi_config.h, frameworks.h
    "${CMAKE_BINARY_DIR}"                 # build-tree ompi/mca/*/base/static-components.h
    "${OMPI_SOURCE_DIR}"
    "${OMPI_SOURCE_DIR}/ompi"             # some sources use "info/..." style
    "${OMPI_SOURCE_DIR}/ompi/include"
    "${CMAKE_BINARY_DIR}/deps-build/hwloc/include"
    "${_hwloc_src}/include"
    "${CMAKE_BINARY_DIR}/deps-build/libevent/include"
    "${_le_src}/include")

target_link_libraries(open-mpi PRIVATE
    open-pal
    event_shared hwloc pmix
    ws2_32 iphlpapi advapi32 dbghelp shlwapi)

set_target_properties(open-mpi PROPERTIES
    OUTPUT_NAME open-mpi
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
